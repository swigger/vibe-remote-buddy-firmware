cmake_minimum_required(VERSION 3.16)
project(vibeled VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(QT NAMES Qt5 Qt6 REQUIRED COMPONENTS Core Widgets SerialPort Network)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Widgets SerialPort Network)
add_library(vibeled_core cli.cpp transport.cpp ../protocol/src/rbp_frame.c)
target_include_directories(vibeled_core PUBLIC . ../protocol/include)
target_link_libraries(vibeled_core PUBLIC Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::SerialPort Qt${QT_VERSION_MAJOR}::Network)
add_executable(vibeled main.cpp window.cpp hotkey.cpp assets/icons.qrc)
target_link_libraries(vibeled PRIVATE vibeled_core Qt${QT_VERSION_MAJOR}::Widgets)
if(APPLE)
    target_sources(vibeled PRIVATE assets/vibeled.icns)
    set_source_files_properties(assets/vibeled.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_target_properties(vibeled PROPERTIES MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_ICON_FILE "vibeled.icns"
        MACOSX_BUNDLE_GUI_IDENTIFIER "com.vibeled.app"
        MACOSX_BUNDLE_BUNDLE_NAME "vibeled"
        MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
        MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
        INSTALL_RPATH "@executable_path/../Frameworks")
    target_link_libraries(vibeled PRIVATE "-framework Carbon")
elseif(WIN32)
    enable_language(RC)
    target_sources(vibeled PRIVATE assets/vibeled.rc)
    target_include_directories(vibeled PRIVATE assets)
    target_link_libraries(vibeled PRIVATE user32)
endif()
if(MSVC)
    target_compile_options(vibeled_core PRIVATE /utf-8 /W4)
    target_compile_options(vibeled PRIVATE /utf-8 /W4)
endif()
install(TARGETS vibeled RUNTIME DESTINATION . BUNDLE DESTINATION .)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
    add_executable(vibeled_test tests.cpp ../firmware/led/led_engine.cpp)
    target_include_directories(vibeled_test PRIVATE ../firmware/led)
    target_link_libraries(vibeled_test PRIVATE vibeled_core Qt${QT_VERSION_MAJOR}::Test)
    if(MSVC)
        target_compile_options(vibeled_test PRIVATE /utf-8)
    endif()
    add_test(NAME vibeled COMMAND vibeled_test)
endif()
